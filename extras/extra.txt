// Function to animate the greeting text
function animateText() {
    const words = document.querySelectorAll('.word');
    words.forEach((word, index) => {
        word.style.animation = 'none';
        word.offsetHeight; // Trigger reflow to reset animation
        word.style.animation = `fadeInWords 2s forwards`;
    });

    const button = document.querySelector('.explore-btn');
    button.style.animation = 'none';
    button.offsetHeight; // Trigger reflow to reset animation
    button.style.animation = 'fadeInBtn 2s 2s forwards';
}

// Function to position words randomly and create connecting lines
function positionWords() {
    const titleWords = document.querySelectorAll('.word');
    const linesContainer = document.querySelector('.lines-container');
    const wordPositions = [];
    
    // Store positions of each word
    titleWords.forEach((word, index) => {
        const randomX = Math.random() * window.innerWidth;
        const randomY = Math.random() * (window.innerHeight / 2); // Restrict random Y position to upper half of the screen
        word.style.left = `${randomX}px`;
        word.style.top = `${randomY}px`;
        
        wordPositions.push({ x: randomX, y: randomY });

        if (index > 0) {
            // Create line connecting the current word to the previous one
            const line = document.createElement('div');
            line.classList.add('line');
            
            const prevWord = wordPositions[index - 1];
            const length = Math.sqrt(Math.pow(randomX - prevWord.x, 2) + Math.pow(randomY - prevWord.y, 2));
            const angle = Math.atan2(randomY - prevWord.y, randomX - prevWord.x) * (180 / Math.PI);

            // Position the line
            line.style.height = `${length}px`;
            line.style.transform = `rotate(${angle}deg)`;
            line.style.left = `${prevWord.x}px`;
            line.style.top = `${prevWord.y}px`;

            linesContainer.appendChild(line);
        }
    });
}

// Initialize the random positions and lines
positionWords();

// Listen for window resize to reposition words and lines
window.addEventListener('resize', positionWords);

function animateText() {
    const title = document.querySelector('.title');
    const subheading = document.querySelector('.subheading');
    const button = document.querySelector('.explore-btn');

    // Triggering animation for greeting text when button is clicked
    title.style.animation = 'none';
    subheading.style.animation = 'none';
    button.style.animation = 'none';

    title.offsetHeight; // Trigger reflow to reset animation
    subheading.offsetHeight; // Trigger reflow to reset animation
    button.offsetHeight; // Trigger reflow to reset animation

    // Re-apply animation after a brief reset
    title.style.animation = 'fadeInText 2s forwards';
    subheading.style.animation = 'fadeInText 2s 1s forwards';
    button.style.animation = 'fadeInBtn 2s 2s forwards';

    // Adding an interactive effect after clicking
    button.textContent = 'Exploring...';
    button.disabled = true;

    setTimeout(() => {
        button.textContent = 'Explore Again';
        button.disabled = false;
    }, 3000);
}
